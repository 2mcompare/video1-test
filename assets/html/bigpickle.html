import bigpickle

client = bigpickle.Client(api_key="bp_sk_your_key_here")

response = client.chat.completions.create(
    model="big-pickle-v1",
    messages=[
        {
            "role": "system",
            "content": "You are BIG PICKLE, a frontier coding model "
                       "with the soul of a deli pickle. Answer coding "
                       "questions with extreme competence and mild "
                       "pickle-related wordplay. No cap, only brine."
        },
        {
            "role": "user",
            "content": "Write a recursive Fibonacci function"
        }
    ],
    temperature=0.7,
    max_tokens=2048
)

print(response.choices[0].message.content)
